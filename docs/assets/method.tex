% Sans for the words, LaTeX maths for everything that is a symbol.
\newcommand{\stagename}[1]{{\sffamily\bfseries\small\color{ink}#1}}
\newcommand{\corename}[1]{{\sffamily\bfseries\small\color{coretitle}#1}}
\newcommand{\note}[1]{{\sffamily\fontsize{6.4}{7.6}\selectfont\color{muted}#1}}
\newcommand{\kicker}[1]{{\sffamily\bfseries\fontsize{6}{7}\selectfont\color{accent}#1}}

% Success as an indicator, which is what it is: the environment's own verdict on
% one episode. Writing it out keeps the retention ratio honest about being a
% ratio of two *means of indicators* rather than of two probabilities.
\newcommand{\succind}[1]{\mathbf{1}\!\left[\,\mathrm{succ}(#1)\,\right]}

% ------------------------------------------------------------------- any model
% The input is the model itself, and the claim is that its *kind* is all xevals
% needs to know. Three of the four protocols, written as the signatures they are.
\node[stage, align=left, inner sep=6pt] (model) at (0, 0) {%
  \stagename{any model}\\[3.5pt]
  $\pi \colon \mathcal{O} \to \mathcal{A}$\\[2.5pt]
  $f \colon \mathcal{O} \times \mathcal{A}^{H} \to \mathcal{P}$\\[2.5pt]
  $g \colon \mathcal{L} \to \mathcal{S}$%
};
\node[below=3.5pt of model, align=center] (modelcap) {%
  \note{policy $\cdot$ world model $\cdot$ planner}\\[1pt]
  \note{torch $\cdot$ jax $\cdot$ LeRobot $\cdot$ HF $\cdot$ HTTP $\cdot$ fn}%
};

% --------------------------------------------------------------------- adapter
\node[stage, right=0.62cm of model] (adapt) {%
  \stagename{adapter}\\[2.5pt]
  $\mathrm{wrap} \colon \mathrm{Model} \to \Pi$%
};
\node[below=4pt of adapt] {\note{numpy at the boundary}};

% ---------------------------------------------------------------------- runner
\node[chip, right=0.62cm of adapt] (pi) {$\pi$};
\node[core, right=0.62cm of pi] (runner) {%
  \corename{runner}\\[2.5pt]
  $\mathrm{roll} \colon (\pi, e, s) \mapsto \tau$%
};
\node[chip, right=0.62cm of runner] (tau) {$\tau$};

\node[input, below=1.86cm of runner] (env) {%
  $e \in \mathcal{E}$\\[1.5pt]
  \note{reset $\cdot$ step $\cdot$ limits}%
};

% The cell grid: five severities across, three families down, the clean column
% marked. It sits between the runner and the environment because it is what the
% runner sweeps rather than a thing in the flow.
\begin{scope}[shift={($(runner.south) + (-0.62, -0.34)$)}]
  \foreach \r in {0, 1, 2} {
    \foreach \c in {0, 1, 2, 3, 4} {
      \ifnum\c=0
        \fill[signal, rounded corners=0.9pt]
          (\c*0.27, -\r*0.20) rectangle ++(0.20, 0.13);
      \else
        \draw[draw=accent, draw opacity=\cellop, line width=0.4pt, rounded corners=0.9pt]
          (\c*0.27, -\r*0.20) rectangle ++(0.20, 0.13);
      \fi
    }
  }
  % Only the clean column is labelled. The SUITE kicker above already names the
  % grid's axes, and the accent column is the one thing a reader has to identify:
  % it is the cell every paired metric in the equation above divides by.
  \node[anchor=north west, inner sep=0pt] at (-0.02, -0.48)
    {{\sffamily\fontsize{5.4}{6}\selectfont\color{signal}clean}};
\end{scope}

% --------------------------------------------------------------------- metrics
\node[stage, right=0.62cm of tau] (metrics) {%
  \stagename{metrics}\\[2.5pt]
  $m \colon \tau^{n} \to (v,\, \mathrm{ci},\, n)$%
};
\node[below=4pt of metrics] {\note{pure functions over trajectories}};

% ------------------------------------------------------------------ dimensions
\node[stage, right=0.62cm of metrics, minimum width=1.45cm] (dims) {%
  \stagename{dimensions}\\[4pt]
  \begin{tikzpicture}[baseline=0pt, x=1cm, y=1cm]
    \foreach \h/\col [count=\i from 0] in
      {0.80/dimA, 0.44/dimB, 0.13/dimC, 0.24/dimD, 1.00/dimE, 1.00/dimF, 1.00/dimG} {
      \fill[\col, rounded corners=0.5pt]
        (\i*0.13, 0) rectangle ++(0.09, {max(0.02, \h*0.34)});
    }
    \draw[draw=accent, draw opacity=0.5, line width=0.35pt] (-0.02, 0) -- (0.93, 0);
  \end{tikzpicture}\\[1.5pt]
  $D_d \in [0, 1]$%
};
\node[below=4pt of dims, align=center] {%
  \note{seven, in a fixed order}%
};

% ------------------------------------------------------------------------ flow

\draw[flow] (pi) -- (runner);
\draw[flow] (runner) -- (tau);
\draw[flow] (tau) -- (metrics);
\draw[flow] (metrics) -- (dims);
\draw[flow] (env) -- (runner);

% The pairing loop: the clean cell's episodes re-enter the metrics as the
% reference every paired metric divides by. This is the only cycle in the
% library, and the equation over it is the only arithmetic worth drawing.
\coordinate (loopmid) at ($(tau.north)!0.5!(metrics.north)$);
\draw[loop] (tau.north) |- ($(loopmid) + (0, 0.86)$) -| (metrics.north);
\node[above=0.90cm of loopmid, anchor=south, align=center] {%
  $\displaystyle \rho_p
     = \frac{\mathbb{E}_{s \in S}\,\succind{p, s}}
            {\mathbb{E}_{s \in S}\,\succind{\varnothing, s}}$\\[2.5pt]
  \note{paired on seeds: the clean cell, the same $S$}%
};

% ---------------------------------------------- what is deliberately absent
\node[align=left, anchor=north west] at ($(model.west |- modelcap.south) + (-0.06, -0.30)$) {%
  \note{$\times$\; no single number: seven dimensions, always}\\[2.5pt]
  \note{$\times$\; nothing unmeasurable scored as $0$: it is $\varnothing$, with a reason}\\[2.5pt]
  \note{$\times$\; no difference claimed that the intervals do not separate}\\[2.5pt]
  \note{$\times$\; no numbers at all when the replay gate fails}%
};

\node[above right=1.5pt and 1pt of zone.north west, anchor=south west]

\node[anchor=north, align=center] at ($(figmid |- zone.south) + (0, -0.52)$) {%

  \note{every $\eta_m$ is written into \texttt{run.json}, so the score can be recomputed, or disagreed with}%
